<script>

const ws = new WebSocket("ws://localhost:8765");


const tremorEl = document.getElementById("tremor");

const hrEl = document.getElementById("hr");

const combinedEl = document.getElementById("combined");

const mlsevEl = document.getElementById("mlsev");

const labelEl = document.getElementById("label");

const ledEl = document.getElementById("led");



const ctx = document.getElementById("chart");


const chart = new Chart(ctx,{

type:'line',

data:{

labels:[],

datasets:[

{

label:'Combined Severity',

data:[],

borderWidth:2

},

{

label:'Heart Rate',

data:[],

borderWidth:2

}

]

},

options:{

animation:false,

responsive:true,

scales:{

y:{

beginAtZero:true

}

}

}

});



ws.onmessage = function(event){

let data = JSON.parse(event.data);


tremorEl.innerText = data.tremor_pct.toFixed(1);

hrEl.innerText = data.hr_bpm.toFixed(1);

combinedEl.innerText = data.combined_pct.toFixed(1);

mlsevEl.innerText = data.ml_severity.toFixed(1);

labelEl.innerText = data.ml_label;

ledEl.innerText = data.led_state;


updateStatusColor(data.ml_label);

updateChart(data);

};



function updateStatusColor(label){

labelEl.className="value status";


if(label=="Normal")

labelEl.classList.add("normal");


else if(label=="Mild")

labelEl.classList.add("mild");


else if(label=="Moderate")

labelEl.classList.add("moderate");


else

labelEl.classList.add("severe");

}



function updateChart(data){

chart.data.labels.push("");


chart.data.datasets[0].data.push(data.combined_pct);

chart.data.datasets[1].data.push(data.hr_bpm);


if(chart.data.labels.length>40){

chart.data.labels.shift();

chart.data.datasets.forEach(d=>d.data.shift());

}


chart.update();

}
